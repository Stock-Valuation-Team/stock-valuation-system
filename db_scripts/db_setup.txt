-- Table: public.holdings

-- DROP TABLE IF EXISTS public.holdings;

CREATE TABLE IF NOT EXISTS public.holdings
(
    symbol text COLLATE pg_catalog."default",
    stock text COLLATE pg_catalog."default",
    qty integer,
    buy_date date,
    buy_price integer,
    exchange text COLLATE pg_catalog."default",
    strategy text COLLATE pg_catalog."default"
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.holdings
    OWNER to postgres;

    id integer NOT NULL DEFAULT nextval('stock_news_id_seq'::regclass),
    ticker text COLLATE pg_catalog."default" NOT NULL,
    title text COLLATE pg_catalog."default",
    publisher text COLLATE pg_catalog."default",
    publish_date timestamp without time zone,
    link text COLLATE pg_catalog."default",
    CONSTRAINT stock_news_pkey PRIMARY KEY (id)
)

    OWNER to postgres;
